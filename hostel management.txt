        function filterComplaints() {
            const catFilter = document.getElementById('complaintCatFilter').value;
            const statusFilter = document.getElementById('complaintStatusFilter').value;

            let filtered = complaintsData.filter(c => {
                const matchCat = (catFilter === 'ALL' || c.category === catFilter);
                const matchStatus = (statusFilter === 'ALL' || c.status === statusFilter);

            });

            renderComplaintsList(filtered);
        }

            studentsData.forEach(st => {
                let roomKey = `${st.mgmtGroup} - ${st.block} - Room ${st.room}`;
                if(!roomMap[roomKey]) {
                    roomMap[roomKey] = {
                        title: roomKey,
                        mgmtGroup: st.mgmtGroup,
                        block: st.block,
                        roomNo: st.room,
                        seaterType: st.seaterType,
                        occupants: []
                    };

                roomMap[roomKey].occupants.push(st);

            let filteredRooms = Object.values(roomMap).filter(r => {
                let matchTitle = r.title.toLowerCase().includes(query);
                let matchOccupant = r.occupants.some(o => o.name.toLowerCase().includes(query));

                container.innerHTML = `<p class="col-span-2 text-center text-xs text-slate-500 py-8">No room allocations match query.</p>`;

        function markAttendance(studentId, newStatus) {
            const previousStatus = attendanceData[studentId];
            attendanceData[studentId] = newStatus;
            renderAttendanceTable();

            if (newStatus === 'Absent' && previousStatus !== 'Absent') {
                triggerParentAbsenceAlert(studentId);
            }
        }

        }

                log.actualReturn = new Date().toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'});
                renderGateLogs();
                showToast(`Marked return for ${log.studentName}`, 'success');
            }
        }

        // ==================== UTILITY FUNCTIONS ====================
        function closeModal(modalId) {
            document.getElementById(modalId).classList.add('hidden');
        }

        function showToast(message, type = 'info') {
            const container = document.getElementById('toastContainer');
            const toast = document.createElement('div');
            let bgClass = type === 'success' ? 'bg-emerald-600 text-white' : 'bg-slate-800 text-white';

            toast.className = `pointer-events-auto px-4 py-2.5 rounded-xl shadow-lg text-xs font-bold flex items-center space-x-2 transition-all transform duration-300 ${bgClass}`;
            toast.innerHTML = `<i class="fa-solid fa-circle-info"></i><span>${message}</span>`;

            container.appendChild(toast);
            setTimeout(() => {
                toast.classList.add('opacity-0', 'translate-y-2');
                setTimeout(() => toast.remove(), 300);
            }, 3500);
        }

            initAttendance();
        };
